
from RPLCD.i2c import CharLCD
from time import sleep

# Initialize the LCD
lcd = CharLCD('PCF8574', 0x27)  # Change 0x27 if needed

# Display the message
lcd.write_string("Hello from Pi 400!")

# Wait for 3 seconds
sleep(3)

# Clear the LCD
lcd.clear()
